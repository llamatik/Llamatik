# Build a portable JNI binary that works on all x86_64 CPUs, not just the build
# machine's microarchitecture. By default ggml sets GGML_NATIVE=ON, which passes
# -march=native to the compiler. GitHub Actions ubuntu-latest runners use Intel
# Xeon CPUs that support AVX-512, so the resulting libllama_jni.so contains
# AVX-512 instructions (e.g. vpternlogd zmm) that execute during ggml_cpu_init —
# before any model loads — and crash with SIGILL on CPUs without AVX-512 support
# (e.g. Intel Tiger Lake i7-1255U, which only has AVX/AVX2).
#
# With GGML_NATIVE=OFF, ggml sets INS_ENB=ON instead, which selects explicit
# compile flags for SSE4.2, AVX, AVX2, FMA, and F16C — a safe x86_64 baseline.
# AVX-512 variants remain OFF (their ggml defaults).
#
# Override for a locally-optimised build:
#   ./gradlew :core:jvmJar -Pllamatik.cmake.args="-DGGML_NATIVE=ON"
if(NOT DEFINED GGML_NATIVE)
    set(GGML_NATIVE OFF CACHE BOOL "ggml: optimize the build for the current system (OFF = portable)" FORCE)
endif()

    # desktop build ggml comes from llama.cpp first so the whisper-side branch is skipped.
    # Keep the Windows fix on the shared ggml-base target used by Java/JNI bindings.
    target_compile_definitions(ggml-base PRIVATE _DISABLE_CONSTEXPR_MUTEX_CONSTRUCTOR)
    message(STATUS "Applied Windows ggml mutex constructor workaround to target: ggml-base")
endif()

    set(SD_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)

    set(SD_BUILD_SHARED_GGML_LIB OFF CACHE BOOL "" FORCE)
    set(SD_USE_SYSTEM_GGML OFF CACHE BOOL "" FORCE)
    # Tell SD to skip patched ggml extensions absent from llama.cpp's upstream ggml.
    set(SD_USE_UPSTREAM_GGML ON CACHE BOOL "" FORCE)

    add_subdirectory("${STABLE_DIFFUSION_CPP_DIR}" "${CMAKE_BINARY_DIR}/stable-diffusion")
    # Provide SD-specific ggml symbols absent from llama.cpp's ggml (same shim as iOS/Android).
    set(SD_GGML_SHIM_DIR "${REPO_ROOT}/core/cmake/llama-wrapper/sd_ggml_shim")
    target_include_directories(stable-diffusion BEFORE PRIVATE "${SD_GGML_SHIM_DIR}")
    target_sources(stable-diffusion PRIVATE "${REPO_ROOT}/core/cmake/llama-wrapper/sd_ggml_shim.cpp")
    if(MSVC)
        # MSVC does not support #include_next (GCC/Clang extension). Generate a thin
        # forwarding header in the build tree that the shim's ggml.h can #include directly.
        set(_ggml_next_dir "${CMAKE_BINARY_DIR}/sd_ggml_shim_msvc")
        file(WRITE "${_ggml_next_dir}/ggml_next.h"
            "#include \"${LLAMA_CPP_DIR}/ggml/include/ggml.h\"\n")
        target_include_directories(stable-diffusion PRIVATE "${_ggml_next_dir}")
    endif()
    set(HAVE_SD ON)
else()
    message(WARNING "stable-diffusion.cpp not found at ${STABLE_DIFFUSION_CPP_DIR}. Desktop will build without SD.")
    add_compile_definitions(LLAMATIK_NO_SD=1)
endif()

    else()
        message(FATAL_ERROR "Stable Diffusion target not found (expected one of: stable-diffusion, stable_diffusion, stable-diffusion-lib).")

    # Force PIC explicitly on the SD static target (some projects ignore global PIC)
    foreach(_t IN ITEMS stable-diffusion stable_diffusion stable-diffusion-lib)
        if(TARGET ${_t})
            set_property(TARGET ${_t} PROPERTY POSITION_INDEPENDENT_CODE ON)
            message(STATUS "Forcing PIC for target: ${_t}")
        endif()
    endforeach()
endif()
